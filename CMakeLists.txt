cmake_minimum_required(VERSION 3.24)
project(Sweepiler VERSION 1.4.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Use Windows x64 with MSVC.")
endif()
include(FetchContent)
FetchContent_Declare(imgui GIT_REPOSITORY https://github.com/ocornut/imgui.git GIT_TAG v1.91.9b GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(imgui)
include(cmake/Z3.cmake)
add_library(imgui STATIC ${imgui_SOURCE_DIR}/imgui.cpp ${imgui_SOURCE_DIR}/imgui_draw.cpp ${imgui_SOURCE_DIR}/imgui_tables.cpp ${imgui_SOURCE_DIR}/imgui_widgets.cpp ${imgui_SOURCE_DIR}/backends/imgui_impl_win32.cpp ${imgui_SOURCE_DIR}/backends/imgui_impl_dx11.cpp)
target_include_directories(imgui PUBLIC ${imgui_SOURCE_DIR} ${imgui_SOURCE_DIR}/backends)
target_link_libraries(imgui PUBLIC d3d11 d3dcompiler dwmapi)
# core: MineC and .imsg compilers, NOR netlists, frame placement and routing, the frame solver and the machine session.
add_library(core STATIC src/Board.cpp src/BuildControl.cpp src/Circuits.cpp src/ClassicCircuits.cpp src/ClassicKernels.cpp src/ComposedCircuits.cpp src/Constraints.cpp src/ClassicSolver.cpp src/CircuitVerification.cpp src/Rules.cpp src/NorNetlist.cpp src/GridLayout.cpp src/MachineFrame.cpp src/FrameSolver.cpp src/FrameMachine.cpp src/GameLanguage.cpp src/MineC.cpp src/MachineSession.cpp)
target_include_directories(core PUBLIC src)
target_link_libraries(core PRIVATE Z3::z3)
target_compile_options(core PRIVATE /W4 /permissive- /utf-8)
# sweepiler: the app, a pixel-art UI on the core (ui/).
add_executable(sweepiler WIN32 ui/LabMain.cpp ui/LabCommon.cpp ui/LabWin32.cpp ui/PixelApp.cpp ui/PixelFont.cpp ui/PixelEditor.cpp ui/PixelDocs.cpp ui/PixelDocsWin32.cpp ui/MineCDocs.cpp)
target_include_directories(sweepiler PRIVATE ui)
target_link_libraries(sweepiler PRIVATE core imgui d3d11 dxgi comdlg32)
target_compile_options(sweepiler PRIVATE /W4 /permissive- /utf-8)
ims_copy_z3(sweepiler)
enable_testing()
function(sweepiler_test name source timeout)
  add_executable(${name}_tests tests/${source})
  target_link_libraries(${name}_tests PRIVATE core)
  target_compile_options(${name}_tests PRIVATE /W4 /permissive- /utf-8)
  ims_copy_z3(${name}_tests)
  add_test(NAME ${name} COMMAND ${name}_tests)
  set_tests_properties(${name} PROPERTIES TIMEOUT ${timeout})
endfunction()
# The ordinary kernels (wire, NOR, crossing ...): correct outputs SAT, wrong outputs UNSAT (Z3).
sweepiler_test(classic_circuits ClassicTests.cpp 120)
sweepiler_test(classic_solver ClassicSolverTests.cpp 120)
sweepiler_test(grid_layout GridLayoutTests.cpp 600)
sweepiler_test(game_language GameLanguageTests.cpp 300)
sweepiler_test(mine_c MineCTests.cpp 600)
sweepiler_test(machine_session MachineSessionTests.cpp 900)
